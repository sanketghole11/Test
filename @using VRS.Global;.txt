        function Insertsalesorderdetails() {
            var user = "Admin";
            var userid = sessionStorage.getItem("LoginName");
            var CoBr_Id = sessionStorage.getItem("CoBr_Id");
            var fcyr = sessionStorage.getItem("FcYr_Id");
            var datax = {
                "designcode": "", "designcode": "", "mrp": "", "size": "", "color": "", "Qty": "", "cobrid": "01", "user": "admin",
            }
            var div = document.getElementById('productContainerAddcard');
            var divs = div.getElementsByTagName('table');
            var tables = [];
            var count = divs.length;
            var runningCount = 0;
            var ajaxRequests = [];

            for (var i = 0; i < divs.length; i += 1) {
                tables = [];
                var table1 = divs[i];
                var table = $('#' + table1.id);
                table.find('tr').each(function (i, el) {
                    var $tds = $(this).find('td');
                    var productId = '';
                    for (var j = 0; j < $tds.length; j++) {
                        if ($tds.eq(j).find("input").val() != undefined) {
                            productId += $tds.eq(j).find("input").val() + ",";
                        } else {
                            productId += $tds.eq(j).text() + ",";
                        }
                    }
                    productId = productId.toString().substring(0, productId.toString().length - 1);
                    tables.push(productId);
                });

                count = tables[1].toString().split(',').length;

                for (var t = 0; t < tables[1].toString().split(',').length; t++) {
                    runningCount = t;
                    //datax.designcode = tables[0].slice(7);
                    datax.designcode = tables[0];
                    datax.mrp = tables[1].toString().split(',')[t + 1];
                    datax.size = tables[2].toString().split(',')[t + 1];

                    for (var z = 3; z < tables.length; z++) {
                        if (tables[z].toString().split(',')[t + 1] != '' && tables[z].toString().split(',')[t + 1] != undefined) {
                            datax.color = tables[z].toString().split(',')[0];
                            datax.Qty = tables[z].toString().split(',')[t + 1];
                            var data = JSON.stringify(datax);
                            ajaxRequests.push($.ajax({
                                type: "POST",
                                url: "/OrderBooking/Insertsalesorderdetails",
                                data: {
                                    userid: userid,
                                    cobr: CoBr_Id,
                                    fcyr: fcyr,
                                    data: data,
                                    typ: '1'
                                }
                            }));
                        }
                    }
                }
            }


            $.when.apply($, ajaxRequests).then(function () {

                if (count == runningCount + 1) {
                    InsertFinalSalesOrder();
                }
                window.location.replace("/OrderRegister/index");
            });
        }

        function stringToColor(str) {
            // Convert the string to a numeric value
            var hash = 0;
            for (var i = 0; i < str.length; i++) {
                hash = str.charCodeAt(i) + ((hash << 5) - hash);
            }
            var color = '#';
            for (var j = 0; j < 3; j++) {
                var value = (hash >> (j * 8)) & 0xFF;
                color += ('00' + value.toString(16)).substr(-2);
            }

            return color;
        }

        function generateProductCard(stylid, imagepath, id, shadeNames, shadeKeys) {
            var x = "";

            // Split shadeNames and shadeKeys into arrays
            var nameArray = (shadeNames || "").split(',').map(s => s.trim()).filter(s => s !== "");
            var keyArray = (shadeKeys || "").split(',').map(s => s.trim()).filter(s => s !== "");

            // Pair names and keys into objects
            var shades = [];
            for (var i = 0; i < nameArray.length; i++) {
                shades.push({ name: nameArray[i], key: keyArray[i] });
            }

            // Remove duplicates by shade name
            var uniqueShades = [];
            var seenNames = new Set();
            for (var i = 0; i < shades.length; i++) {
                if (!seenNames.has(shades[i].name)) {
                    seenNames.add(shades[i].name);
                    uniqueShades.push(shades[i]);
                }
            }

            // Generate Shade Buttons
            var shadeButtons = '<div class="container-fluid">';
            for (var i = 0; i < uniqueShades.length; i++) {
                if (i % 5 === 0) {
                    if (i !== 0) shadeButtons += '</div>';
                    shadeButtons += '<div class="row justify-content-start mb-2 px-3">';
                }
                shadeButtons += `
                    <div class="col-md-2 col-sm-4 col-6 mb-2 d-flex justify-content-center">
                        <button
                            id="colorBtn${stylid}_${i}"
                            class="btn btn-outline-primary w-100 shade-button"
                            style="font-size: 14px; font-weight: normal;"
                            data-shadename="${uniqueShades[i].name}"
                            data-shadekey="${uniqueShades[i].key}"
                            data-stylid="${stylid}"
                            onclick="toggleShadeSelection(this)">
                            ${uniqueShades[i].name}
                        </button>
                        <input type="hidden" id="shadeKey${stylid}_${i}" value="${uniqueShades[i].key}">
                    </div>
                `;
            }
            shadeButtons += '</div></div>';

            // Generate Shade Rows inside table
            var dynamicShadeRows = "";
            for (var j = 0; j < uniqueShades.length; j++) {
                dynamicShadeRows += `
                    <tr id="row-${stylid}_${j}">
                        <th scope="row">${uniqueShades[j].name}</th>
                        <td colspan="6">
                            <input type="text" class="form-control" id="shadeInput${stylid}_${j}" placeholder="Enter quantity">
                            <input type="hidden" id="shadeKeyHidden${stylid}_${j}" value="${uniqueShades[j].key}">
                        </td>
                    </tr>
                `;
            }

            // Full Product Card HTML
            x = `
                <div id="pc${stylid}" class="product-card col-md-12 col-lg-12 col-xl-6 mb-4">
                    <div class="content padding-all-15">
                        <div class="card text-left zoom shadow-sm">

                            <!-- Shade Buttons -->
                            <div class="p-3">
                                ${shadeButtons}
                            </div>

                            <!-- Product Image -->
                            <div class="text-center">
                                <input type="checkbox" id="checkbox${stylid}" class="product-checkbox" data-item-id="${stylid}">
                                <img id="img${stylid}" class="img-fluid card-img-top p-2" src="${imagepath}" alt="Product Image">
                            </div>

                            <!-- Collapsible Product Details -->
                            <div class="collapse" id="collapsedetails${stylid}">
                                <div class="card-body card-show">
                                    <div class="white_bg padding-all-1 overflow-auto">
                                        <table id="tbl${stylid}" class="table table-bordered table-sm">
                                            <thead>
                                                <tr>
                                                    <th scope="col">Design No:</th>
                                                    <th scope="col">C00001</th>
                                                    <th colspan="5"></th>
                                                </tr>
                                            </thead>
                                            <tbody>
                                                <tr>
                                                    <th scope="row">MRP</th>
                                                    <td colspan="6"><label>1500</label></td>
                                                </tr>
                                                <tr>
                                                    <th scope="row">Size</th>
                                                    <td>All</td><td>30</td><td>28</td><td>25</td><td>22</td><td>20</td>
                                                </tr>
                                                <tr><td colspan="7" class="fw-bold">Shades</td></tr>
                                                ${dynamicShadeRows}
                                            </tbody>
                                        </table>
                                        <div class="text-center">
                                            <a onclick="Addcart(this,0)" id="Addcart${stylid}" class="btn btn-primary">Add</a>
                                        </div>
                                    </div>
                                </div>
                            </div>

                            <!-- Footer Buttons -->
                            <div id="crd${id}" class="card-footer text-center">
                                <a class="btn btn-danger me-2" id="bk${stylid}" data-toggle="collapse" onclick="BookNow(this)" role="button" style="padding: 10px 20px; font-size: 16px;">
                                    Book Now
                                </a>
                                <a onclick="AddcartFav(this,0)" id="bk${stylid}" class="btn btn-primary" style="padding: 10px 20px; font-size: 16px;">
                                    Favorites
                                </a>
                            </div>

                        </div>
                    </div>
                </div>
            `;

            return x;
        }

            var cobr = sessionStorage.getItem("CoBr_Id");
            var fcyr = sessionStorage.getItem("FcYr_Id");

                url: "/OrderBooking/GetOrderDetailsHeader",
                data: {
                    itemSubGrpKey: selectedSubGrp,
                    ItemKey: selectedItem,
                    brandKey: selectedBrand,
                    styleKey: selectedStyle,
                    shadeKey: selectedShade,
                    sizeKey: selectedSize,
                    stockWise: stockWise,
                    fromMRP: fromMRP,
                    toMRP: toMRP,
                    userid: userid,
                    cobr: cobr,
                    fcyr: fcyr

                success: function (response) {

                        alert("No records found.");
                        return;
                    }

                    // Handle success response here
                    console.log(response);
                },

                    // Handle error here
                    console.log(error);
                }
            });
        }

        function showTooltip(input) {
            var tooltip = document.createElement('div');
            tooltip.className = 'custom-tooltip';
            tooltip.innerHTML = input.getAttribute('title');
            document.body.appendChild(tooltip);
            var rect = input.getBoundingClientRect();
            tooltip.style.top = rect.top + window.scrollY - tooltip.offsetHeight + 'px';
            tooltip.style.left = rect.left + window.scrollX + 'px';
        }
        function hideTooltip(input) {
            var tooltips = document.querySelectorAll('.custom-tooltip');
            tooltips.forEach(function (tooltip) {
                tooltip.remove();
            });
        }

        function calculateTotalQty(input, styleCode) {
            var totalQty = 0;
            var totQtyId = 'totQty' + styleCode;
            document.getElementById(totQtyId).value = totalQty;
            var inputFields = input.closest('table').querySelectorAll('.form-control');
            inputFields.forEach(function (inputField) {
                if (!isNaN(inputField.value) && inputField.value !== '') {
                    totalQty += parseFloat(inputField.value);
                    inputField.style.fontWeight = 'bold';
                } else {
                    inputField.style.fontWeight = 'normal';
                }
            });
            document.getElementById(totQtyId).value = totalQty;
        }

                "creditdays": $('#creditdays').val(), "createddate": doc_dt,
            };
            var data = JSON.stringify(datax);

            $.ajax({
                type: "POST",
                url: "/OrderBooking/InsertCust",
                data: {
                    data: data,
                    cobr: cobr,
                    fcyr: fcyr,
                    user: userid
                },
                success: function (response) {
                    if (response === "0") {
                        alert("Customer already exists with this name.");
                    } else if (response === "1") {
                        alert("Customer saved successfully.");
                        $('#addcust').modal('hide');
                        getCustomer('W');
                    } else {
                        alert("An error occurred while saving the customer.");
                    }
                },
                error: function (xhr, status, error) {
                    alert("Customer Not saved successfully. Error: " + error);
                }
            });
        }
